"""Host-specific sandbox for the fixed actor; test launch cleanup never touches old runners."""
import json
import os
import select
import subprocess
import sys
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
ACTOR = HERE / "actor"
PIN = Path(sys.executable).resolve().parents[2]


def profile():
    # Read system dependencies, but not user, mounted-volume or temporary data.
    # Explicit actor/runtime exceptions contain source only, never actual-world data.
    exceptions = "(allow file-read* (subpath " + json.dumps(str(PIN)) + ")) "
    exceptions += "(allow file-read* (literal " + json.dumps(str(ACTOR)) + ")) "
    exceptions += " ".join("(allow file-read* (literal " + json.dumps(str(ACTOR / name)) + "))"
                           for name in ("worker.py", "wire.py", "compute.py", "selector.py"))
    outside = " ".join("(require-not (subpath " + json.dumps(p) + "))"
                       for p in ("/Users", "/Volumes", "/private/tmp", "/private/var/folders", "/tmp"))
    return ("(version 1) (deny default) (allow process-exec) (allow sysctl-read) "
            "(allow mach-lookup) (allow file-read-metadata) " + exceptions +
            " (allow file-read* (require-all " + outside + "))")


def command(script=None, code=None):
    if sys.platform != "darwin" or not Path("/usr/bin/sandbox-exec").exists():
        raise RuntimeError("validated_macOS_sandbox_required_no_fallback")
    if (script is None) == (code is None):
        raise ValueError("exactly_one_entry")
    return ["/usr/bin/sandbox-exec", "-p", profile(), sys.executable, "-I", "-B", "-S",
            *([str(script)] if script is not None else ["-c", code])]


ENV = {"PATH": "/usr/bin:/bin", "LC_ALL": "C", "PYTHONDONTWRITEBYTECODE": "1"}


class Actor:
    def __init__(self):
        self.started = time.monotonic()
        self.p = subprocess.Popen(command(script=ACTOR / "worker.py"), stdin=subprocess.PIPE,
                                  stdout=subprocess.PIPE, stderr=subprocess.PIPE, bufsize=0,
                                  close_fds=True, cwd="/", env=ENV)
        self.buffer = bytearray()
        self.exited = False
        self.forced_cleanup = False

    def send(self, value):
        data = json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode() + b"\n"
        self.p.stdin.write(data)
        self.p.stdin.flush()

    def receive(self, timeout=30):
        deadline = time.monotonic() + timeout
        while b"\n" not in self.buffer:
            remaining = deadline - time.monotonic()
            if remaining <= 0 or not select.select([self.p.stdout], [], [], max(0, remaining))[0]:
                raise TimeoutError("synthetic_actor_timeout")
            block = os.read(self.p.stdout.fileno(), 65536)
            if not block:
                detail = b""
                if select.select([self.p.stderr], [], [], 0)[0]:
                    detail = os.read(self.p.stderr.fileno(), 2048)
                raise RuntimeError("actor_eof_without_complete_result:" + detail.decode(errors="replace")[:700])
            self.buffer.extend(block)
            if len(self.buffer) > 131072:
                raise RuntimeError("actor_output_budget")
        line, _, rest = self.buffer.partition(b"\n")
        self.buffer = bytearray(rest)
        value = json.loads(line)
        if value.get("schema") == "actor-error-v1":
            raise RuntimeError("actor_rejected:" + value["error_code"])
        return value

    def finish(self):
        self.p.stdin.close()
        self.p.wait(timeout=5)
        extra = bytes(self.buffer) + self.p.stdout.read(131073)
        stderr = self.p.stderr.read(8193)
        self.exited = True
        self.p.stdout.close()
        self.p.stderr.close()
        if self.p.returncode != 0 or extra or stderr:
            raise RuntimeError("actor_exit_or_extra_output")
        return {"pid": self.p.pid, "exit_code": self.p.returncode, "exited": True,
                "forced_cleanup": False, "process_wall_seconds": time.monotonic() - self.started}

    def close(self):
        if not self.exited:
            if self.p.stdin and not self.p.stdin.closed:
                self.p.stdin.close()
            try:
                self.p.wait(timeout=3)
            except subprocess.TimeoutExpired:
                # Only the newly created test child owned by this Actor instance.
                self.forced_cleanup = True
                self.p.kill()
                self.p.wait(timeout=3)
            self.exited = True
            for stream in (self.p.stdout, self.p.stderr):
                if stream and not stream.closed:
                    stream.close()
