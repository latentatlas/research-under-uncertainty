"""Restricted generated policy evaluator; no file/network capabilities in policy globals."""
import ast
import cmath
import json
import math
import resource
import sys

FORBIDDEN = (ast.ClassDef, ast.Global, ast.Nonlocal, ast.AsyncFunctionDef,
             ast.Await, ast.Yield, ast.YieldFrom)
BUILTINS = {name: value for name, value in {
    'abs': abs, 'min': min, 'max': max, 'sum': sum, 'len': len, 'range': range,
    'enumerate': enumerate, 'zip': zip, 'list': list, 'tuple': tuple, 'dict': dict,
    'str': str, 'set': set, 'float': float, 'int': int, 'complex': complex, 'bool': bool,
    'round': round, 'sorted': sorted, 'reversed': reversed, 'all': all, 'any': any,
    'isinstance': isinstance, 'map': map, 'filter': filter, 'pow': pow,
    'Exception': Exception, 'ValueError': ValueError, 'ZeroDivisionError': ZeroDivisionError,
}.items()}

def safe_import(name, globals=None, locals=None, fromlist=(), level=0):
    if level or name not in ('math', 'cmath'):
        raise ValueError('only math/cmath imports permitted')
    return {'math': math, 'cmath': cmath}[name]

def compile_policy(code):
    if not isinstance(code, str) or len(code) > 60000:
        raise ValueError('policy code missing or too long')
    tree = ast.parse(code)
    for node in ast.walk(tree):
        if isinstance(node, FORBIDDEN): raise ValueError('unsupported statement')
        if isinstance(node, ast.Attribute) and node.attr.startswith('_'):
            raise ValueError('private attributes forbidden')
        if isinstance(node, ast.Name) and node.id.startswith('__'):
            raise ValueError('private names forbidden')
        if isinstance(node, (ast.FunctionDef, ast.arg)):
            name = node.name if isinstance(node, ast.FunctionDef) else node.arg
            if name.startswith('__'): raise ValueError('private bindings forbidden')
        if isinstance(node, ast.Import) and any(x.name not in ('math','cmath') for x in node.names):
            raise ValueError('only math/cmath imports permitted')
        if isinstance(node, ast.ImportFrom):
            if node.level or node.module not in ('math','cmath') or any(x.name.startswith('_') for x in node.names):
                raise ValueError('only public math/cmath imports permitted')
    compiled = compile(tree, '<submitted-policy>', 'exec')
    def policy(history):
        namespace = {'__builtins__': dict(BUILTINS, __import__=safe_import)}
        exec(compiled, namespace, namespace)
        if not callable(namespace.get('act')): raise ValueError('act(history) not defined')
        return namespace['act'](history)
    return policy
