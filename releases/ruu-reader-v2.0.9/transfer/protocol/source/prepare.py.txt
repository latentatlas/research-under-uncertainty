"""Generate the fixed prospective inputs once, without any model calls."""
from copy import deepcopy
import cmath,math,hashlib,os
from local import HERE,ROOT,save,read,sha,digest,now
from tasks import task
from continuous import physics as W
from continuous import snapshot_core as C
import route
from wire import validate_payload

SEED='transfer-validation-20261025-v1'
PROFILES=[{'model':m,'reasoning_effort':'max','mode':'single_agent'} for m in ('gpt-6-astra','gpt-6.1-sol')]

def generate():
    cases=[]
    for i in range(1,4):
        raw=hashlib.sha256((SEED+':'+str(i)).encode()).digest()
        w={'world_id':'transfer-continuous-'+str(i),'phi0':2*math.pi*int.from_bytes(raw[:4],'big')/2**32,
           'omega':.08+.36*int.from_bytes(raw[4:8],'big')/2**32,'coupling':(3.,6.,12.)[i-1],
           'mechanism_version':'rotating_core_kappa_axis_v1'}
        e=W.Episode(w);snapshots={}
        for k,r in enumerate((1.1,1.2,1.3,1.4,1.5,1.6,1.75,1.9,2.05,2.2),1):
            target=r*cmath.exp(1j*.08*math.sin(k+i))
            a=(target-W.drift(e.z))*cmath.exp(-1j*e.phi)
            e.step([a.real,a.imag])
            if k in (6,10):snapshots[k]=e.history()
        for index,k in enumerate((6,10),1):
            cases.append({'family':'continuous','instance':i,'state_index':index,'configuration':w,'history':snapshots[k],'step':k})
        rw={'instance':i,'base':[2+raw[8]%2,3+raw[9]%2,5+raw[10]%2],
            'offset':[raw[11+j]%4 for j in range(3)],'period':[2+raw[14+j]%3 for j in range(3)]}
        table=route.value_table(rw);rw['initial_fuel']=table[12][0]+1
        h={'states':[[0,rw['initial_fuel']]],'actions':[]}
        for k in range(1,5):
            a=route.best_action(rw,h['states'][-1][0],13-k,table)
            h['actions'].append(a);h['states'].append(route.transition(rw,h['states'][-1],a))
            if k in (2,4):cases.append({'family':'route','instance':i,'state_index':k//2,'configuration':rw,'history':deepcopy(h),'step':k})
    return sorted(cases,key=lambda c:(c['instance'],c['state_index'],c['family']))

def main():
    os.umask(0o077)
    if (HERE/'CASES.private.json').exists() or (HERE/'FREEZE.json').exists():raise FileExistsError('prepared already')
    rows=[]
    for i,c in enumerate(generate(),1):
        t=task(c['family']);cid=f'snapshot-{i:02d}'
        p={'case_id':cid,'history':c.pop('history'),'physical_step':c.pop('step'),'task':t}
        p['remaining_physical_steps']=t['physical_horizon']-p['physical_step']
        validate_payload({'snapshot':p,'analysis_records':[],'remaining_analysis_calls':3,'remaining_model_decisions':4})
        path=f'inputs.private/{cid}.json';save(HERE/path,p)
        rows.append(c|{'case_id':cid,'delivery_path':path,'delivery_sha256':sha(HERE/path),'history_sha256':digest(p['history'])})
    save(HERE/'CASES.private.json',rows)
    allocation=[]
    # Separate repeats in time; reverse profile order across cases and repeats.
    for rep in (1,2):
        for i,c in enumerate(rows):
            for profile in (PROFILES if (i+rep)%2==1 else PROFILES[::-1]):
                allocation.append({k:c[k] for k in ('case_id','family','instance','state_index','delivery_path','delivery_sha256')}|
                    {'ordinal':len(allocation)+1,'repeat':rep,'profile':profile,'packet_sha256':c['delivery_sha256']})
    save(HERE/'ALLOCATION.json',allocation)
    limits={'max_calls':192,'per_call_timeout_seconds':900,'max_input_tokens':201600000,'max_output_tokens':24576000,'max_elapsed_seconds':3780}
    save(HERE/'PLAN.json',{'schema':'prospective-transfer-panel-v1','created_utc':now(),'design_sha256':sha(HERE/'DESIGN.md'),
        'seed':SEED,'families':2,'instances_per_family':3,'snapshots_per_instance':2,'profiles':PROFILES,'repeats':2,
        'planned_trials':48,'max_model_requests':192,'limits':limits,'per_call_reservation':{'input':1050000,'output':128000},
        'neutrals':0,'retries':0,'old_h4_resume':False,'new_family':'finite_horizon_route_fuel',
        'primary':'fixed_denominator_diagnostic_coverage','secondary':['repeat_category_agreement','frozen_table_adapter_applicability','record_based_crossings'],
        'development_trials_excluded':8,'calibrated_reliability':False,'universal_generality':False,'publication_authorized':False})
    files=['grading.py','route.py','tasks.py','components.py','CASES.private.json','ALLOCATION.json']+[str(p.relative_to(HERE)) for p in (HERE/'continuous').glob('*.py')]+[c['delivery_path'] for c in rows]
    save(HERE/'GRADING-LOCK.json',{'files':{f:sha(HERE/f) for f in files},'semantics':'family-specific positive witnesses and valid loss certificates; unresolved is not loss'})
    print({'cases':len(rows),'allocations':len(allocation),'model_calls':0})

if __name__=='__main__':main()
