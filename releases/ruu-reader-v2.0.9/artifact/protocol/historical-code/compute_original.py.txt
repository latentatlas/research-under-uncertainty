"""A capability-limited numerical tool; only supplied public history enters act."""
import json
from pathlib import Path
import resource
import sys
from policy_runtime import compile_policy

def compute(code,history):
    try:
        result=compile_policy(code)(history)
        encoded=json.dumps(result,allow_nan=False)
        if len(encoded)>100000: raise ValueError('calculation output exceeds 100000 bytes')
        return {'status':'ok','value':result}
    except Exception as exc:
        return {'status':'code_error','error':type(exc).__name__+': '+str(exc)}

if __name__=='__main__':
    resource.setrlimit(resource.RLIMIT_CPU,(2,3))
    req=json.load(sys.stdin)
    print(json.dumps(compute(req['code'],req['history']),allow_nan=False))
