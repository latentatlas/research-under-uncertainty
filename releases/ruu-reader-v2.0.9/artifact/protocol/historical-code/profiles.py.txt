"""Explicit participant configurations; a profile is never silently downgraded."""
from __future__ import annotations

from dataclasses import asdict, dataclass

from strict_io import digest


EXECUTABLE_PROFILES = frozenset({
    ("gpt-6-astra", "max"),
    ("gpt-6.1-sol", "max"),
})


@dataclass(frozen=True)
class ParticipantProfile:
    model: str
    reasoning_effort: str
    mode: str = "single_agent"

    def __post_init__(self):
        if any(not isinstance(value, str) for value in (self.model, self.reasoning_effort, self.mode)):
            raise ValueError("Profile fields must be strings")
        if self.mode == "single_agent":
            if (self.model, self.reasoning_effort) not in EXECUTABLE_PROFILES:
                raise ValueError("Unregistered single-agent profile; no automatic fallback")
        else:
            raise ValueError("Unknown participant mode")

    def to_dict(self):
        return asdict(self)

    @classmethod
    def from_dict(cls, value):
        if not isinstance(value, dict) or set(value) != {"model", "reasoning_effort", "mode"}:
            raise ValueError("Profile requires exactly model, reasoning_effort and mode")
        if any(not isinstance(item, str) for item in value.values()):
            raise ValueError("Profile fields must be strings")
        return cls(**value)

    @property
    def fingerprint(self):
        return digest(self.to_dict())

    def require_executable(self):
        if self.mode != "single_agent":
            raise ValueError("Astra/ultra is a planned multi-agent condition; this transport cannot execute it")


@dataclass(frozen=True)
class ResourceLimits:
    max_calls: int
    per_call_timeout_seconds: int
    max_input_tokens: int
    max_output_tokens: int
    max_elapsed_seconds: int

    def __post_init__(self):
        if any(type(value) is not int or value < 1 for value in asdict(self).values()):
            raise ValueError("Every resource limit must be a positive integer")

    def to_dict(self):
        return asdict(self)

    @classmethod
    def from_dict(cls, value):
        if not isinstance(value, dict) or set(value) != set(cls.__dataclass_fields__):
            raise ValueError("Resource limits require exactly the five registered fields")
        return cls(**value)
