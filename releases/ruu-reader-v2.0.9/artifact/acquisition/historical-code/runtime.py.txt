USAGE = {"synthetic_history_steps": 0, "synthetic_extra_observations": 0,
         "synthetic_truth_steps": 0, "actor_processes": 0, "forced_actor_cleanup": 0,
         "main_history_steps": 0, "main_extra_observations": 0, "main_truth_steps": 0}


def write_new(path, value):
    data = canonical(value) + b"\n"
    with Path(path).open("xb") as f:
        f.write(data)
        f.flush()
        os.fsync(f.fileno())
    return hashlib.sha256(data).hexdigest()


def read(path):
    return json.loads(Path(path).read_bytes())


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def validate_table(table):
    need(type(table) in (list, tuple) and len(table) in (2, 4, 6), "environment_table_shape")
    states = len(table) // 2
    need(all(type(edge) in (list, tuple) and len(edge) == 2 and type(edge[0]) is int
             and 0 <= edge[0] < states and bit(edge[1]) for edge in table), "environment_edge")
    return tuple(tuple(e) for e in table)


class Environment:
    """Trusted environment role. Its private state is never serialized to the actor."""
    def __init__(self, trial_id, table, actions, *, main=False):
        need(type(main) is bool and trial_id.startswith("t" if main else "s"), "environment_scope_identity")
        self.__usage_prefix = "main_" if main else "synthetic_"
        self.__table = validate_table(table)
        need(type(actions) in (list, tuple) and len(actions) == 8 and all(bit(a) for a in actions),
             "environment_history_actions")
        state = 0
        history = []
        for action in actions:
            state, output = self.__table[2 * state + action]
            history.append([action, output])
            USAGE[self.__usage_prefix + "history_steps"] += 1
        self.__cutoff = state
        self.__original_digest = digest([self.__table, state, history])
        self.__history = copy.deepcopy(history)
        self.__view = make_view(trial_id, history)
        self.__clones = {arm: {"state": state, "used": 0} for arm in ARMS}
        self.__selections = {}
        self.__truth_read = False
        self.work = {"environment_resets": 1, "history_generation_steps": 8,
                     "environment_clones": 2, "environment_replay_steps": 0,
                     "extra_observations": {arm: 0 for arm in ARMS}, "truth_steps": 0}

    def permitted_view(self):
        return copy.deepcopy(self.__view)

    def original_digest(self):
        return digest([self.__table, self.__cutoff, self.__history])

    def observe(self, selection):
        need(type(selection) is dict, "selection_fields")
        view = self.permitted_view()
        validate_selection(selection, view, selection.get("arm"))
        need(self.original_digest() == self.__original_digest, "original_state_changed")
        arm = selection["arm"]
        need(self.__clones[arm]["used"] == 0, "one_observation_budget")
        need(self.__clones[arm]["state"] == self.__cutoff, "clone_cutoff_changed")
        need(self.__clones["fixed"] is not self.__clones["adaptive"], "shared_arm_state")
        self.__selections[arm] = digest(selection)
        clone = self.__clones[arm]
        clone["state"], output = self.__table[2 * clone["state"] + (selection["query_id"] == "q1")]
        clone["used"] += 1
        self.work["extra_observations"][arm] += 1
        USAGE[self.__usage_prefix + "extra_observations"] += 1
        need(self.original_digest() == self.__original_digest, "observation_mutated_original")
        return {"schema": "one-step-observation-v1", "trial_id": view["trial_id"],
                "arm": arm, "view_sha256": digest(view), "query_id": selection["query_id"],
                "output": output}

    def reveal_after_commit(self, ledger, permit):
        ledger.check_permit(permit, self.permitted_view())
        need(not self.__truth_read, "duplicate_truth_read")
        need(self.original_digest() == self.__original_digest, "truth_original_state_changed")
        values = []
        # Privileged evaluator deliberately follows the two step paths; it does not
        # import the candidate target function or use either arm's changed state.
        for first_action in (0, 1):
            state = self.__cutoff
            for action in (first_action, 0):
                state, output = self.__table[2 * state + action]
                self.work["truth_steps"] += 1
                USAGE[self.__usage_prefix + "truth_steps"] += 1
            values.append(output)
        self.__truth_read = True
        ledger.note_truth(self.permitted_view())
        return 2 * values[0] + values[1]


def validate_selection(selection, view, arm):
    need(arm in ARMS, "selection_arm")
    fields = {"schema", "trial_id", "arm", "view_sha256", "query_id",
              "pre_observation_history_checks", "pre_observation_query_evaluations", "possible_gain_loss"}
    need(type(selection) is dict and set(selection) == fields, "selection_fields")
    need(selection["schema"] == "query-selection-v1", "selection_schema")
    need(selection["trial_id"] == view["trial_id"] and selection["arm"] == arm
         and selection["view_sha256"] == digest(view), "selection_identity")
    need(selection["query_id"] in QUERIES, "selection_query")
    for key in ("pre_observation_history_checks", "pre_observation_query_evaluations"):
        need(type(selection[key]) is int and selection[key] >= 0, "selection_work")
    if arm == "fixed":
        need(selection["query_id"] == "q0" and selection["pre_observation_history_checks"] == 0
             and selection["pre_observation_query_evaluations"] == 0
             and selection["possible_gain_loss"] is None, "fixed_pre_observation_contract")
    else:
        need(selection["pre_observation_history_checks"] == 46656 and
             0 < selection["pre_observation_query_evaluations"] <= 93312, "adaptive_pre_observation_work")
        p = selection["possible_gain_loss"]
        need(type(p) is dict and set(p) == {"gain_possible", "loss_possible"}
             and all(type(v) is bool for v in p.values()), "possible_gain_loss_fields")


def validate_work(result, selection):
    work = result["work"]
    need(type(work) is dict and set(work) == set(COUNTERS), "work_fields_missing_or_extra")
    need(all(type(v) is int and v >= 0 for v in work.values()), "invalid_work_counter")
    c = result["compatible_before"]
    adaptive = result["arm"] == "adaptive"
    need(work["candidate_decodes"] == work["history_checks"] == work["candidate_history_resets"] == 46656,
         "one_full_history_pass")
    need(46656 <= work["history_steps"] <= 46656 * 8, "history_work_bound")
    need(work["target_candidate_evaluations"] == c and work["target_candidate_steps"] == 4 * c,
         "target_work_identity")
    need(work["query_candidate_evaluations"] == work["query_candidate_steps"] == c * (2 if adaptive else 1),
         "query_work_identity")
    need(work["selection_branch_checks"] == (4 if adaptive else 0), "selection_work_identity")
    need(work["certificate_branch_lookups"] == 1 and
         work["within_arm_reused_query_values"] == (c if adaptive else 0), "reuse_work_identity")
    if adaptive:
        need(selection["pre_observation_query_evaluations"] == work["query_candidate_evaluations"],
             "pre_observation_meter_identity")
    t = result["timing"]
    need(type(t) is dict and set(t) == {"actor_cpu_seconds", "actor_wall_seconds"}
         and all(type(v) in (int, float) and math.isfinite(v) and v >= 0 for v in t.values()), "timing_missing_or_invalid")


def validate_arm(bundle, view, arm):
    need(type(bundle) is dict and set(bundle) == {"selection", "observation", "result", "process"}, "arm_bundle_fields")
    sel, obs, result, process = (bundle[k] for k in ("selection", "observation", "result", "process"))
    validate_selection(sel, view, arm)
    guard_receipt(obs, view, arm, sel["query_id"])
    fields = {"schema", "trial_id", "arm", "view_sha256", "selection_sha256", "observation_sha256",
              "query_id", "output", "certificate", "prediction", "target_values", "compatible_after",
              "compatible_before", "initial_target_cardinality", "chosen_branch_target_cardinality",
              "work", "timing", "pid", "process_io_gate"}
    need(type(result) is dict and set(result) == fields, "result_fields")
    need(result["schema"] == "arm-result-v1" and result["trial_id"] == view["trial_id"] and result["arm"] == arm
         and result["view_sha256"] == digest(view), "result_identity")
    need(result["selection_sha256"] == digest(sel) and result["observation_sha256"] == digest(obs), "result_chain")
    need(result["query_id"] == obs["query_id"] and result["output"] == obs["output"], "result_observation")
    for k in ("compatible_before", "compatible_after"):
        need(type(result[k]) is int and 0 < result[k] <= 46656, "candidate_count")
    need(result["compatible_after"] <= result["compatible_before"], "support_increase")
    targets = result["target_values"]
    need(type(targets) is list and 1 <= len(targets) <= 4 and
         all(type(x) is int and 0 <= x <= 3 for x in targets) and targets == sorted(set(targets)), "target_set")
    need(type(result["initial_target_cardinality"]) is int and len(targets) <= result["initial_target_cardinality"] <= 4,
         "initial_target_cardinality")
    need(type(result["chosen_branch_target_cardinality"]) is int and
         result["chosen_branch_target_cardinality"] == len(targets), "branch_cardinality")
    need(type(result["certificate"]) is bool and result["certificate"] == (len(targets) == 1), "certificate_rule")
    need(result["prediction"] == (targets[0] if len(targets) == 1 else None) and
         (result["prediction"] is None or type(result["prediction"]) is int), "prediction_rule")
    need(type(process) is dict and set(process) == {"pid", "exit_code", "exited", "forced_cleanup", "process_wall_seconds"},
         "process_receipt_fields")
    need(type(result["pid"]) is int and result["pid"] > 0 and process["pid"] == result["pid"] and
         process["exit_code"] == 0 and process["exited"] is True and process["forced_cleanup"] is False,
         "actor_process_not_cleanly_exited")
    need(result["process_io_gate"] == "OS_read_scope_plus_post_load_Python_audit_hook", "actor_access_gate")
    need(type(process["process_wall_seconds"]) in (int, float) and math.isfinite(process["process_wall_seconds"])
         and process["process_wall_seconds"] >= 0, "process_timing")
    validate_work(result, sel)


def validate_pair(pair, view):
    need(type(pair) is dict and set(pair) == {"schema", "view", "arms", "environment_work", "original_unchanged"}, "pair_fields")
    need(pair["schema"] == "paired-observation-result-v1" and canonical(pair["view"]) == canonical(view), "pair_identity")
    guard_view(pair["view"])
    need(set(pair["arms"]) == set(ARMS), "paired_arms_required")
    for arm in ARMS:
        validate_arm(pair["arms"][arm], view, arm)
    f, a = (pair["arms"][arm]["result"] for arm in ARMS)
    need(f["initial_target_cardinality"] == a["initial_target_cardinality"] and
         f["compatible_before"] == a["compatible_before"], "same_history_different_support")
    if a["query_id"] == "q0":
        for k in ("output", "certificate", "prediction", "target_values", "compatible_after"):
            need(f[k] == a[k], "same_query_different_scientific_output")
    ew = pair["environment_work"]
    need(type(ew) is dict and set(ew) == {"environment_resets", "history_generation_steps", "environment_clones",
         "environment_replay_steps", "extra_observations", "truth_steps"}, "environment_meter_fields")
    need(ew == {"environment_resets": 1, "history_generation_steps": 8, "environment_clones": 2,
                "environment_replay_steps": 0, "extra_observations": {"fixed": 1, "adaptive": 1}, "truth_steps": 0},
         "environment_work_or_early_truth_violation")
    need(pair["original_unchanged"] is True, "original_state_changed")


def collect_pair(environment, order=ARMS):
    need(tuple(order) in (ARMS, tuple(reversed(ARMS))), "arm_order")
    original = environment.original_digest()
    view = environment.permitted_view()
    arms = {}
    for arm in order:
        actor = Actor()
        USAGE["actor_processes"] += 1
        try:
            actor.send({"arm": arm, "view": view})
            selection = actor.receive()
            validate_selection(selection, view, arm)
            observation = environment.observe(selection)
            actor.send(observation)
            result = actor.receive()
            process = actor.finish()
            arms[arm] = {"selection": selection, "observation": observation, "result": result, "process": process}
            validate_arm(arms[arm], view, arm)
        finally:
            actor.close()
            USAGE["forced_actor_cleanup"] += actor.forced_cleanup
    pair = {"schema": "paired-observation-result-v1", "view": view, "arms": arms,
            "environment_work": copy.deepcopy(environment.work),
            "original_unchanged": environment.original_digest() == original}
    validate_pair(pair, view)
    return pair


def scientific_projection(pair):
    result = copy.deepcopy(pair)
    for bundle in result["arms"].values():
        bundle.pop("process")
        for key in ("timing", "pid"):
            bundle["result"].pop(key)
    return result


class Ledger:
    """Trusted commit authority. Hashes detect changes; privileged host remains trusted."""
    def __init__(self, directory, planned_views, source_manifest_sha256, *, main_binding=None):
        self.directory = Path(directory)
        self.directory.mkdir(exist_ok=False)
        need(0 < len(planned_views) <= 5000, "positive_planned_count")
        self.views = {v["trial_id"]: guard_view(copy.deepcopy(v)) for v in planned_views}
        need(len(self.views) == len(planned_views), "duplicate_trial_identity")
        self.main = main_binding is not None
        if self.main:
            need(type(main_binding) is dict and set(main_binding) ==
                 {"N", "freeze_sha256", "authorization_sha256", "draw_sha256"}, "main_binding_fields")
            need(type(main_binding["N"]) is int and main_binding["N"] == len(self.views) == 5000,
                 "fixed_main_N5000")
            need(list(self.views) == ["t%04d" % i for i in range(1, 5001)], "main_ordered_identity")
            need(all(type(main_binding[k]) is str and len(main_binding[k]) == 64
                     and all(c in "0123456789abcdef" for c in main_binding[k])
                     for k in ("freeze_sha256", "authorization_sha256", "draw_sha256")), "main_binding_hashes")
            need(source_manifest_sha256 == main_binding["draw_sha256"], "main_draw_binding")
        else:
            need(all(t.startswith("s") for t in self.views), "synthetic_dispatch_only")
        self.N = len(planned_views)
        self.phase = "collecting"
        self._permit = None
        self._truth_ids = set()
        self.rows = {}
        self.events = []
        self.plan = {"schema": "paired-main-plan-v1" if self.main else "synthetic-control-plan-v1",
                     "kind": "fixed_N_main_study" if self.main else "synthetic_controls",
                     "planned_N": self.N, "planned_ids": list(self.views),
                     "views": {t: digest(v) for t, v in self.views.items()},
                     "source_manifest_sha256": source_manifest_sha256}
        if self.main:
            self.plan["main_binding"] = copy.deepcopy(main_binding)
        write_new(self.directory / "PLAN.json", self.plan)
        self.plan_sha256 = sha(self.directory / "PLAN.json")

    def commit(self, pair):
        need(self.phase == "collecting", "commit_after_seal_forbidden")
        trial = pair["view"]["trial_id"]
        need(trial in self.views and trial not in self.rows, "commit_trial_identity")
        validate_pair(pair, self.views[trial])
        path = self.directory / (trial + ".json")
        h = write_new(path, pair)
        path.chmod(0o400)
        self.rows[trial] = h
        self.events.append({"event": "pair_committed", "trial_id": trial, "index": len(self.events)})

    def seal(self):
        need(self.phase == "collecting" and list(self.rows) == list(self.views), "all_planned_pairs_required_in_order")
        seal = {"schema": "all-pairs-commitment-v1", "planned_N": self.N,
                "plan_sha256": self.plan_sha256, "ordered_rows": list(self.rows.items()),
                "truth_reads_before_commitment": len(self._truth_ids)}
        need(not self._truth_ids, "early_truth_read")
        write_new(self.directory / "COMMITMENT.json", seal)
        self.seal_sha256 = sha(self.directory / "COMMITMENT.json")
        (self.directory / "COMMITMENT.json").chmod(0o400)
        self.phase = "sealed"
        self.events.append({"event": "all_pairs_sealed", "index": len(self.events)})
        self.verify()
        return self.seal_sha256

    def verify(self):
        need(self.phase in ("sealed", "auditing", "closed"), "all_commitments_required_before_truth")
        need(sha(self.directory / "PLAN.json") == self.plan_sha256, "plan_hash_mismatch")
        need(sha(self.directory / "COMMITMENT.json") == self.seal_sha256, "commitment_hash_mismatch")
        seal = read(self.directory / "COMMITMENT.json")
        need(seal["plan_sha256"] == self.plan_sha256 and seal["planned_N"] == self.N and
             seal["ordered_rows"] == [[t, self.rows[t]] for t in self.views] and
             seal["truth_reads_before_commitment"] == 0, "commitment_chain")
        for t, expected in self.rows.items():
            need(sha(self.directory / (t + ".json")) == expected, "sealed_row_hash_mismatch")
            validate_pair(read(self.directory / (t + ".json")), self.views[t])
        return True

    def authorize_truth(self):
        need(self.phase == "sealed", "truth_phase")
        self.verify()
        self.phase = "auditing"
        self._permit = object()
        return self._permit

    def check_permit(self, permit, view):
        need(self.phase == "auditing" and self._permit is not None and permit is self._permit, "truth_before_complete_commitment")
        trial = view["trial_id"]
        need(trial in self.views and digest(view) == digest(self.views[trial]), "truth_identity")
        need(trial not in self._truth_ids, "truth_already_read")

    def note_truth(self, view):
        trial = view["trial_id"]
        self._truth_ids.add(trial)
        self.events.append({"event": "truth_read", "trial_id": trial, "index": len(self.events)})

    def incomplete(self):
        need(self.phase == "collecting" and len(self.rows) < self.N, "incomplete_state")
        cells = dict.fromkeys(("n11", "n10", "n01", "n00"), 0)
        for t in self.rows:
            pair = read(self.directory / (t + ".json"))
            a = pair["arms"]["adaptive"]["result"]["certificate"]
            f = pair["arms"]["fixed"]["result"]["certificate"]
            cells["n%d%d" % (a, f)] += 1
        missing = self.N - len(self.rows)
        return {"planned_N": self.N, "completed_pairs": len(self.rows), "missing_pairs": missing,
                "issued_cells_not_truth_validated": cells, "truth_reads": 0,
                "decision": statistics.decision(self.N, **cells, technical_missing=missing),
                "replacement_draws": 0, "automatic_retry": False}


def readout(ledger, environments):
    need(list(environments) == list(ledger.views), "truth_environment_identity_order")
    permit = ledger.authorize_truth()
    cells = dict.fromkeys(("n11", "n10", "n01", "n00"), 0)
    totals = {arm: dict.fromkeys(COUNTERS, 0) for arm in ARMS}
    timings = {arm: dict.fromkeys(("actor_cpu_seconds", "actor_wall_seconds", "process_wall_seconds"), 0.0) for arm in ARMS}
    diagnostics = {"initially_ambiguous": 0, "initially_singleton": 0, "query_disagreement": 0,
                   "possible_gain": 0, "possible_loss": 0, "realized_gain_with_possible_gain": 0,
                   "realized_loss_with_possible_loss": 0}
    branch_hits = {arm: {"all_draws": 0, "initially_ambiguous": 0, "query_disagreement": 0} for arm in ARMS}
    wrong_certificates = wrong_pairs = issued = 0
    truth_rows = []
    for t in ledger.views:
        pair = read(ledger.directory / (t + ".json"))
        truth = environments[t].reveal_after_commit(ledger, permit)
        f, a = (pair["arms"][arm]["result"] for arm in ARMS)
        ac, fc = a["certificate"], f["certificate"]
        cells["n%d%d" % (ac, fc)] += 1
        ambiguous = a["initial_target_cardinality"] > 1
        different = a["query_id"] != f["query_id"]
        p = pair["arms"]["adaptive"]["selection"]["possible_gain_loss"]
        diagnostics["initially_ambiguous"] += ambiguous
        diagnostics["initially_singleton"] += not ambiguous
        diagnostics["query_disagreement"] += different
        diagnostics["possible_gain"] += p["gain_possible"]
        diagnostics["possible_loss"] += p["loss_possible"]
        diagnostics["realized_gain_with_possible_gain"] += ac and not fc and p["gain_possible"]
        diagnostics["realized_loss_with_possible_loss"] += fc and not ac and p["loss_possible"]
        row_wrong = 0
        for arm, result in zip(ARMS, (f, a)):
            issued += result["certificate"]
            wrong = result["certificate"] and result["prediction"] != truth
            row_wrong += wrong
            branch_hits[arm]["all_draws"] += result["certificate"]
            branch_hits[arm]["initially_ambiguous"] += result["certificate"] and ambiguous
            branch_hits[arm]["query_disagreement"] += result["certificate"] and different
            for k in COUNTERS:
                totals[arm][k] += result["work"][k]
            for k, v in result["timing"].items():
                timings[arm][k] += v
            timings[arm]["process_wall_seconds"] += pair["arms"][arm]["process"]["process_wall_seconds"]
        wrong_certificates += row_wrong
        wrong_pairs += row_wrong > 0
        truth_rows.append({"trial_id": t, "truth": truth, "wrong_certificates": row_wrong})
    statistic_start = time.process_time()
    decision = statistics.decision(ledger.N, **cells, wrong_certificates=wrong_pairs)
    stats_cpu = time.process_time() - statistic_start
    net = cells["n10"] - cells["n01"]
    marginal = {k: totals["adaptive"][k] - totals["fixed"][k] for k in COUNTERS}
    marginal_timing = {k: timings["adaptive"][k] - timings["fixed"][k] for k in timings["fixed"]}
    denominators = {"all_draws": ledger.N, "initially_ambiguous": diagnostics["initially_ambiguous"],
                    "query_disagreement": diagnostics["query_disagreement"]}
    rate_table = {arm: {key: {"numerator": value, "denominator": denominators[key],
                             "rate": value / denominators[key] if denominators[key] else None}
                        for key, value in group.items()} for arm, group in branch_hits.items()}
    result = {"schema": "paired-observation-readout-v1", "kind": "fixed_N_main_study" if ledger.main else "synthetic_controls",
              "scientific_population_sample": ledger.main, "planned_N": ledger.N, "completed_pairs": ledger.N,
              "technical_missing": 0, "unstarted": 0, "unknown_work_records": 0,
              "cells": cells, "net_certificates": net, "issued_certificates": issued,
              "wrong_certificates": wrong_certificates, "wrong_pairs": wrong_pairs,
              "certificate_audit": "not_assessed_no_certificates" if not issued else
              "invalid_certificate_hold" if wrong_certificates else
              "all_issued_match_on_this_study_sample" if ledger.main else "all_issued_match_on_this_control_set",
              "decision": decision, "diagnostics": diagnostics, "branch_hit_rates": rate_table,
              "work": totals, "timing": timings, "marginal_work": marginal,
              "marginal_work_per_net_certificate": {k: statistics.extra_work_per_net_certificate(v, cells["n10"], cells["n01"])
                                                    for k, v in marginal.items()},
              "marginal_timing": marginal_timing,
              "marginal_timing_per_net_certificate": {k: v / net if net > 0 else None for k, v in marginal_timing.items()},
              "shared_environment_work": {"history_generation_steps": 8 * ledger.N, "environment_resets": ledger.N,
                                          "environment_clones": 2 * ledger.N, "environment_replay_steps": 0,
                                          "extra_observations": 2 * ledger.N, "truth_audit_steps": 4 * ledger.N,
                                          "certificate_truth_comparisons": issued, "statistics_cpu_seconds": stats_cpu},
              "economic_utility": "unassessed", "main_population_draws": ledger.N if ledger.main else 0, "provider_calls": 0}
    ledger.verify()
    write_new(ledger.directory / "PRIVATE-TRUTH-AUDIT.json", truth_rows)
    write_new(ledger.directory / "EVENTS.json", ledger.events)
    write_new(ledger.directory / "READOUT.json", result)
    ledger.phase = "closed"
    write_new(ledger.directory / "CLOSURE.json", {"status": "passed" if not wrong_certificates else "invalid_certificate_hold",
              "commitment_sha256": ledger.seal_sha256,
              "truth_audit_sha256": sha(ledger.directory / "PRIVATE-TRUTH-AUDIT.json"),
              "readout_sha256": sha(ledger.directory / "READOUT.json"),
              "events_sha256": sha(ledger.directory / "EVENTS.json"), "all_actor_children_exited": True})
    return result
