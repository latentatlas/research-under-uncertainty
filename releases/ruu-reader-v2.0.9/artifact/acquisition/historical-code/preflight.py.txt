"""Integration controls use only abstract fixtures/synthetic frames, never main draws."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent))
import copy
import json
import tempfile
import runtime as r
import sampling
import study
import fixtures
import audit_saved
from wire import make_view, digest, need


def run():
    target = study.HERE / "qa/INTEGRATION.json"
    need(not target.exists(), "one_integration_record")
    checks = []
    def check(name, value):
        need(value, name); checks.append(name)
    def reject(name, fn):
        try:
            fn()
        except (ValueError, RuntimeError):
            checks.append(name); return
        raise AssertionError(name)
    previous = audit_saved.audit_checks(study.HERE / "qa/preflight-v01")
    check("all153_runtime_controls_saved_arithmetic", previous["checks"] == 153)
    # Abstract population=7 is deliberately unrelated to the real sampling frame.
    limit = 2**32 // 7 * 7
    s = sampling.draw(7, 3, [limit, 2**32-1, 0, 0, 8])
    check("rejection_boundary_and_duplicate_retention", s["ranks"] == [0,0,1] and len(s["receipts"]) == 5)
    check("equal_modular_multiplicity_below_limit", limit % 7 == 0 and 0 <= 2**32-limit < 7)
    check("entropy_exhaustion_is_incomplete", sampling.draw(7,2,[1])["status"] == "technical_entropy_incomplete")
    check("deterministic_seed_stream", list(sampling.words("00"*32))[:24] == list(sampling.words("00"*32))[:24])
    check("different_seed_stream", list(sampling.words("00"*32))[:24] != list(sampling.words("01"*32))[:24])
    for word in [True, -1, 2**32, 1.0]:
        reject("invalid_entropy_"+repr(word), lambda word=word: sampling.draw(7,1,[word]))
    split_pairs = [sampling.split(i,3) for i in range(768)]
    check("exhaustive_synthetic_pair_mapping", len({(i,tuple(a)) for i,a in split_pairs}) == 768)
    check("history_bit_order", sampling.split(257,3) == (1,[0,0,0,0,0,0,0,1]))
    fixture = fixtures.definitions()[0]
    with tempfile.TemporaryDirectory(prefix="paired-main-preflight-") as temporary:
        tmp = Path(temporary)
        # Same production collector, but no main frame/source or main IDs are used here.
        class FakeJournal:
            def __init__(self): self.rows=[]
            def add(self,event,**kw): self.rows.append({"event":event,**kw})
        import time
        env = r.Environment(fixture["trial_id"], fixture["table"], fixture["actions"])
        direct_env = r.Environment(fixture["trial_id"], fixture["table"], fixture["actions"])
        journal = FakeJournal(); receipts = tmp / "stage-receipts"; receipts.mkdir()
        pair = study.collect(env,receipts,journal,time.monotonic()+120)
        direct = r.collect_pair(direct_env)
        check("main_collector_scientific_equivalence_on_abstract_fixture", r.scientific_projection(pair) == r.scientific_projection(direct))
        check("main_collector_process_exit_journal", [x["event"] for x in journal.rows] == ["actor_started","actor_exited"]*2)
        for arm in r.ARMS:
            for key,name in [("selection","SELECTION.json"),("observation","OBSERVATION.json"),("result","RESULT.json"),("process","PROCESS.json")]:
                check("persisted_stage_"+arm+"_"+key, r.read(receipts/arm/name) == pair["arms"][arm][key])
        # Test main plan identities with fabricated public views: no sampled worlds or observations.
        views = [make_view("t%04d"%i, pair["view"]["history"]) for i in range(1,5001)]
        grant = {"N":5000,"freeze_sha256":"a"*64,"authorization_sha256":"b"*64,"draw_sha256":"c"*64}
        ledger = r.Ledger(tmp/"main-plan", views, "c"*64, main_binding=grant)
        check("full_main_plan_accepted_without_execution", ledger.N==5000 and ledger.main and len(ledger.rows)==0)
        reject("main_short_N_denied", lambda:r.Ledger(tmp/"short",views[:1],"c"*64,main_binding=grant))
        reject("main_order_denied", lambda:r.Ledger(tmp/"reversed",list(reversed(views)),"c"*64,main_binding=grant))
        reject("main_without_binding_denied",lambda:r.Ledger(tmp/"unbound",views,"c"*64))
        reject("main_wrong_draw_binding_denied",lambda:r.Ledger(tmp/"draw-mismatch",views,"d"*64,main_binding=grant))
        bad = dict(grant,freeze_sha256="not-a-hash")
        reject("malformed_main_binding_denied",lambda:r.Ledger(tmp/"bad-hash",views,"c"*64,main_binding=bad))
        reject("main_truth_before_all_pairs_denied",lambda:ledger.authorize_truth())
        reject("main_partial_plan_seal_denied",lambda:ledger.seal())
        check("main_incomplete_retains5000",ledger.incomplete()["missing_pairs"]==5000)
        reject("scope_mismatched_environment_denied",lambda:r.Environment("t0001",fixture["table"],fixture["actions"]))
        # Preserve synthetic receipt evidence before temporary cleanup.
        artifact = {"pair":pair,"direct_projection":r.scientific_projection(direct),"journal":journal.rows,
                    "synthetic_main_plan_sha256":r.sha(tmp/"main-plan/PLAN.json")}
        r.write_new(study.HERE/"qa/PRIVATE-INTEGRATION-EVIDENCE.json",artifact)
    check("temporary_controls_cleaned",not Path(temporary).exists())
    check("no_main_draw_or_observation_or_truth",not study.RUN.exists() and
          r.USAGE["main_history_steps"]==r.USAGE["main_extra_observations"]==r.USAGE["main_truth_steps"]==0)
    check("actor_sources_unchanged",all(r.sha(study.HERE/name)==r.sha(study.PRIOR_RUNTIME/name)
          for name in ("actor/worker.py","actor/compute.py","actor/wire.py","actor/selector.py","paired_statistics.py","sandbox.py")))
    r.write_new(target,{"status":"passed","checks":checks,"count":len(checks),"source_hashes":study.sources(),
        "synthetic_usage":dict(r.USAGE),"synthetic_main_plan_views":5000,"real_population_draws":0,
        "provider_calls":0,"all_actor_children_exited":True,
        "evidence_sha256":r.sha(study.HERE/"qa/PRIVATE-INTEGRATION-EVIDENCE.json")})
    print(json.dumps({"status":"passed","integration_controls":len(checks),"main_draws":0,"main_observations":0,
                      "synthetic_usage":r.USAGE}))


if __name__=="__main__": run()
