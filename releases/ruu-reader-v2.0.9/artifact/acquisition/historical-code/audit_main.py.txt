"""Read-only main-study identities, saved arithmetic and work audit; no world replay."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent))
import json
from collections import Counter
import runtime as r
import study
import sampling
import audit_saved
from wire import digest, need


def audit():
    f = study.validate_freeze()
    run = study.RUN
    need(not (run / "FAILURE.json").exists(), "failed_main_attempt_preserved_no_success_claim")
    c = r.read(run / "CLOSURE.json")
    need(c["status"] in ("passed", "invalid_certificate_hold") and c["all_actor_children_exited"] is True,
         "main_closure")
    paths = {"freeze_sha256": study.HERE / "FREEZE.json", "authorization_sha256": study.HERE / "AUTHORIZATION.json",
             "draw_sha256": run / "PRIVATE-DRAW.json", "source_bindings_sha256": run / "PRIVATE-SOURCE-BINDINGS.json",
             "start_sha256": run / "START.json", "dispatch_sha256": run / "DISPATCH.json",
             "records_closure_sha256": run / "records/CLOSURE.json", "usage_sha256": run / "USAGE.json",
             "journal_sha256": run / "JOURNAL.jsonl"}
    for key, path in paths.items():
        need(r.sha(path) == c[key], "main_closure_chain:" + key)
    d = r.read(run / "PRIVATE-DRAW.json")
    reconstructed = sampling.draw(2143232, 5000, sampling.words(r.read(study.HERE / "PRIVATE-SEED.json")["seed_hex"]))
    for key, value in reconstructed.items():
        need(d[key] == value, "seeded_mapping_reconstruction")
    for key in ("seed_sha256", "eligibility_sha256", "catalog_sha256"):
        need(d[key] == f[key], "draw_source_identity")
    frame, eligible, catalog = study.source_frame()
    source_rows = r.read(run / "PRIVATE-SOURCE-BINDINGS.json")
    ids = ["t%04d" % i for i in range(1, 5001)]
    plan = r.read(run / "records/PLAN.json")
    need(plan["planned_ids"] == ids and plan["planned_N"] == 5000 and plan["kind"] == "fixed_N_main_study",
         "main_full_ordered_plan")
    binding = {"N": 5000, "freeze_sha256": r.sha(study.HERE / "FREEZE.json"),
               "authorization_sha256": f["authorization_sha256"], "draw_sha256": r.sha(run / "PRIVATE-DRAW.json")}
    need(plan["main_binding"] == binding and plan["source_manifest_sha256"] == binding["draw_sha256"], "plan_dispatch_binding")
    need(len(source_rows) == len(d["ranks"]) == 5000, "source_count")
    audit_saved.audit_run(run / "records", c["status"])
    result = r.read(run / "records/READOUT.json")
    timings = {a: {k: 0.0 for k in result["timing"][a]} for a in r.ARMS}
    source_classes = []
    for t, rank, source in zip(ids, d["ranks"], source_rows):
        ei, actions = sampling.split(rank, 8372)
        si = eligible["eligible_source_local_indices"][ei]
        need(source["trial_id"] == t and source["rank"] == rank and source["eligible_index"] == ei
             and source["source_index"] == si and source["original_index"] == catalog["original_indices"][si]
             and source["table_sha256"] == digest(catalog["worlds"][si])
             and source["actions_sha256"] == digest(actions), "per_draw_source_identity")
        source_classes.append(si)
        pair = r.read(run / "records" / (t + ".json"))
        need([a for a, _ in pair["view"]["history"]] == actions
             and source["view_sha256"] == digest(pair["view"]) == plan["views"][t], "draw_history_binding")
        start = r.read(run / "receipts" / t / "START.json")
        need(start["trial_id"] == t and start["view_sha256"] == plan["views"][t], "record_start_identity")
        for arm in r.ARMS:
            for key, filename in [("selection", "SELECTION.json"), ("observation", "OBSERVATION.json"),
                                  ("result", "RESULT.json"), ("process", "PROCESS.json")]:
                need(pair["arms"][arm][key] == r.read(run / "receipts" / t / arm / filename), "stage_receipt_binding")
            for key, value in pair["arms"][arm]["result"]["timing"].items():
                timings[arm][key] += value
            timings[arm]["process_wall_seconds"] += pair["arms"][arm]["process"]["process_wall_seconds"]
    need(timings == result["timing"], "timing_totals")
    net = result["net_certificates"]
    marginal_timing = {k: timings["adaptive"][k] - timings["fixed"][k] for k in timings["fixed"]}
    need(marginal_timing == result["marginal_timing"] and
         result["marginal_timing_per_net_certificate"] == {k: v/net if net>0 else None for k,v in marginal_timing.items()},
         "timing_net_ratio")
    journal = [json.loads(line) for line in (run / "JOURNAL.jsonl").read_text().splitlines()]
    need([x["index"] for x in journal] == list(range(len(journal))), "journal_order")
    events = Counter(x["event"] for x in journal)
    need(events == {"dispatch":1, "all_draws_and_views_bound":1, "pair_started":5000,
        "actor_started":10000, "actor_exited":10000, "pair_committed":5000,
        "collection_complete":1, "all_predictions_sealed_before_truth":1, "truth_audit_complete":1, "closed":1},
        "journal_event_inventory")
    need(journal[0]["event"] == "dispatch" and journal[1]["event"] == "all_draws_and_views_bound", "journal_start")
    expected_events = ["pair_started", "actor_started", "actor_exited", "actor_started", "actor_exited", "pair_committed"]
    for i, t in enumerate(ids):
        group = journal[2 + i*6:2 + (i+1)*6]
        need([e["event"] for e in group] == expected_events and all(e["trial_id"] == t for e in group), "journal_pair_order")
        for j, arm in [(1,"fixed"),(3,"adaptive")]:
            a,b = group[j:j+2]
            process = r.read(run / "receipts" / t / arm / "PROCESS.json")
            need(a["arm"] == b["arm"] == arm and a["pid"] == b["pid"] == process["pid"] and
                 b["exit_code"] == 0 and b["forced_cleanup"] is False, "journal_process_exit")
    need([e["event"] for e in journal[-4:]] == ["collection_complete", "all_predictions_sealed_before_truth", "truth_audit_complete", "closed"],
         "journal_commit_truth_order")
    u = r.read(run / "USAGE.json")
    need(u["main_usage"] == {"synthetic_history_steps":0,"synthetic_extra_observations":0,"synthetic_truth_steps":0,
         "actor_processes":10000,"forced_actor_cleanup":0,"main_history_steps":40000,
         "main_extra_observations":10000,"main_truth_steps":20000}, "main_actual_usage")
    need(u["unaccounted_main_observations"] == u["unknown_actor_work_records"] == u["provider_calls"] == 0, "main_usage_known")
    baseline = r.read(study.HERE / "BASELINE.json")["files"]
    for rel, h in baseline.items():
        need(r.sha(study.ROOT / rel) == h, "old_immutable_file_changed:" + rel)
    return {"status":"passed", "kind":"saved_main_evidence_audit_without_world_replay", "planned_N":5000,
        "completed_pairs":5000,"unstarted":0,"technical_missing":0,"unknown_work_records":0,
        "source_identity_checks":5000,"arm_stage_receipts_checked":40000,
        "all_actor_children_exited":True,"actor_processes":10000,"forced_cleanup":0,
        "selected_prior_files_preserved":len(baseline),"readout_sha256":r.sha(run / "records/READOUT.json"),
        "main_closure_sha256":r.sha(run / "CLOSURE.json"),"source_freeze_sha256":r.sha(study.HERE / "FREEZE.json"),
        "distinct_class_history_draws":len(set(d["ranks"])),"duplicate_draw_excess":5000-len(set(d["ranks"])),
        "distinct_classes_drawn":len(set(source_classes)),"entropy_words":len(d["receipts"]),
        "rejected_entropy_words":sum(not x["accepted"] for x in d["receipts"]),
        "new_draws_in_this_audit":0,"world_execution_in_this_audit":0,"provider_calls":0,
        "scientific_decision":result["decision"],"wrong_certificates":result["wrong_certificates"],
        "independent_laboratory":False}


if __name__ == "__main__":
    value = audit()
    if sys.argv[1:] == ["--record"]:
        r.write_new(study.HERE / "AUDIT.json", value)
    elif sys.argv[1:]:
        raise SystemExit("usage: audit_main.py [--record]")
    print(json.dumps(value, indent=2))
